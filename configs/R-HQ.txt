!======================================================================
! R-HQ - Head office router (Cisco 2911)
! REFERENCE SCRIPT: the commands used for this project.
! Replace this file with the output of "show running-config" from
! your own device before publishing.
! Serial ports: the HWIC module in this lab is in slot 3, so the serial
! ports are s0/3/0 and s0/3/1.
! (Not yet replaced with the real export from the device.)
!======================================================================
enable
configure terminal
hostname R-HQ
!
! --- Access lists ---
access-list 1 permit 192.168.0.0 0.0.255.255
ip access-list extended GUEST-IN
 permit ip 192.168.30.0 0.0.0.255 host 192.168.30.1
 deny ip 192.168.30.0 0.0.0.255 192.168.0.0 0.0.255.255
 permit ip any any
 exit
!
! --- LAN: router-on-a-stick ---
interface gi0/0
 description TRUNK to SW-HQ1
 no shutdown
 exit
interface gi0/0.10
 description STAFF VLAN 10
 encapsulation dot1Q 10
 ip address 192.168.10.1 255.255.255.0
 ip nat inside
 exit
interface gi0/0.20
 description SERVERS VLAN 20
 encapsulation dot1Q 20
 ip address 192.168.20.1 255.255.255.0
 ip nat inside
 exit
interface gi0/0.30
 description GUEST VLAN 30
 encapsulation dot1Q 30
 ip address 192.168.30.1 255.255.255.0
 ip nat inside
 ip access-group GUEST-IN in
 exit
interface gi0/0.99
 description MGMT VLAN 99 (native)
 encapsulation dot1Q 99 native
 ip address 192.168.99.1 255.255.255.0
 exit
!
! --- WAN to branches (DCE end is here) ---
interface s0/3/0
 description WAN to R-B1
 ip address 10.0.12.1 255.255.255.252
 clock rate 64000
 ip nat inside
 no shutdown
 exit
interface s0/3/1
 description WAN to R-B2
 ip address 10.0.13.1 255.255.255.252
 clock rate 64000
 ip nat inside
 no shutdown
 exit
!
! --- Link to ISP ---
interface gi0/1
 description to R-ISP (NAT outside)
 ip address 203.0.113.2 255.255.255.252
 ip nat outside
 no shutdown
 exit
!
! --- Routing ---
ip route 0.0.0.0 0.0.0.0 203.0.113.1
router ospf 1
 router-id 1.1.1.1
 passive-interface default
 no passive-interface s0/3/0
 no passive-interface s0/3/1
 network 192.168.10.0 0.0.0.255 area 0
 network 192.168.20.0 0.0.0.255 area 0
 network 192.168.30.0 0.0.0.255 area 0
 network 192.168.99.0 0.0.0.255 area 0
 network 10.0.12.0 0.0.0.3 area 0
 network 10.0.13.0 0.0.0.3 area 0
 default-information originate
 exit
!
! --- DHCP ---
ip dhcp excluded-address 192.168.10.1 192.168.10.10
ip dhcp excluded-address 192.168.30.1 192.168.30.10
ip dhcp pool STAFF
 network 192.168.10.0 255.255.255.0
 default-router 192.168.10.1
 dns-server 8.8.8.8
 exit
ip dhcp pool GUEST
 network 192.168.30.0 255.255.255.0
 default-router 192.168.30.1
 dns-server 8.8.8.8
 exit
!
! --- NAT ---
ip nat inside source list 1 interface gi0/1 overload
!
! --- Device hardening: SSH only (lab-only passwords) ---
enable secret Cisco@123
service password-encryption
ip domain-name lab.local
username admin secret Admin@123
crypto key generate rsa modulus 2048
ip ssh version 2
line vty 0 4
 transport input ssh
 login local
 exit
line console 0
 password Cisco@123
 login
 end
copy running-config startup-config
